cmake_minimum_required(VERSION 3.20)
project(MonchiLatencyTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(latency_control latency_control.cpp)
target_include_directories(latency_control PRIVATE ../../dll/src)
add_test(NAME latency_control COMMAND latency_control)

add_executable(gpu_probe gpu_probe.cpp ../../dll/src/system/GpuLatency.cpp)
target_include_directories(gpu_probe PRIVATE ../../dll/src)
target_compile_definitions(gpu_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(gpu_probe PRIVATE d3d11 d3d12 dxgi)
if(MSVC)
    target_compile_options(latency_control PRIVATE /W4 /utf-8)
    target_compile_options(gpu_probe PRIVATE /W4 /utf-8)
endif()
